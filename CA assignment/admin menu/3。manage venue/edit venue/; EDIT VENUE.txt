; EDIT VENUE
; ==============================

EDIT_VENUE:

	
    ; Clear screen before displaying this function/page
    MOV AH,00H
    MOV AL,03H
    INT 10H

	MOV AH,09H
    LEA DX,NEWLINE
    INT 21H
	
    MOV AH,09H
    LEA DX,editTitle
    INT 21H

    ; CHECK IF ANY VENUE EXISTS
    CMP VENUE1_EXIST,1
    JE EDIT_HAS_VENUE

    CMP VENUE2_EXIST,1
    JE EDIT_HAS_VENUE

    CMP VENUE3_EXIST,1
    JE EDIT_HAS_VENUE

    MOV AH,09H
    LEA DX,noVenueMsg
    INT 21H
    JMP MANAGE_VENUE


EDIT_HAS_VENUE:

    ; SHOW VENUE NUMBER + NAME
    CALL DISPLAY_VENUE_LIST

    MOV AH,09H
    LEA DX,editMsg
    INT 21H

    MOV AH,01H
    INT 21H


    CMP AL,'1'
    JNE EDIT_CHECK_2

    CMP VENUE1_EXIST,1
    JE EDIT1_OK

    JMP EDIT_NOT_FOUND


EDIT1_OK:

    CALL SET_VENUE1
    CALL INPUT_VENUE
    JMP EDIT_SUCCESS


EDIT_CHECK_2:

    CMP AL,'2'
    JNE EDIT_CHECK_3

    CMP VENUE2_EXIST,1
    JE EDIT2_OK

    JMP EDIT_NOT_FOUND


EDIT2_OK:

    CALL SET_VENUE2
    CALL INPUT_VENUE
    JMP EDIT_SUCCESS


EDIT_CHECK_3:

    CMP AL,'3'
    JNE EDIT_INVALID

    CMP VENUE3_EXIST,1
    JE EDIT3_OK

    JMP EDIT_NOT_FOUND


EDIT3_OK:

    CALL SET_VENUE3
    CALL INPUT_VENUE


EDIT_SUCCESS:

    MOV AH,09H
    LEA DX,venueUpdatedMsg
    INT 21H

    JMP MANAGE_VENUE


EDIT_NOT_FOUND:

    CALL SHOW_NOT_FOUND
    JMP EDIT_VENUE


EDIT_INVALID:

    CALL SHOW_INVALID
    JMP EDIT_VENUE
